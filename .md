```python
def test_draw_by_reserve_exhaustion():
    g = OtrioGame()
    b = g.getInitBoard()
    p = 1
    for _ in range(18):                 # 2人×9枚＝18手で駒切れ
        a = np.where(g.getValidMoves(b,p)==1)[0][0]
        b, p = g.getNextState(b, p, a)
    assert g.getGameEnded(b, p) == 1e-4
```

```python
def test_color_toggle():
    g = OtrioGame()
    b = g.getInitBoard(); p=1
    assert g.next_color[0] == 0
    b,_ = g.getNextState(b,p,0)
    assert g.next_color[0] == 1
    b,_ = g.getNextState(b,-p,9)  # 相手番、小色0
    assert g.next_color[1] == 1
```
